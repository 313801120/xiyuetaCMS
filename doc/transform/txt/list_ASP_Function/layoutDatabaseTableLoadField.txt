function layoutDatabaseTableLoadField()
    dim thisConn:Set thisConn = CreateObject("Adodb.Connection")
    dim sql,fieldList,splstr,s,nOK,temp_db_PREFIX,sortrank,fieldAlt,idIndexName
    temp_db_PREFIX=db_PREFIX
    nOK=0
    sql="select * from ["& db_PREFIX &"layout_database_table] where id="&id
    rs.open sql,conn,1,3
    if not rs.eof then
        sql="select * from ["& db_PREFIX &"layout_database] where id="&rs("layout_database_id")
        rsx.open sql,conn,1,1
        if not rsx.eof then

            temp_db_PREFIX=db_PREFIX 
            if rsx("databaseType")="sqlserver" or rsx("databaseType")="mysql" then
                db_PREFIX=""
                databaseType = rsx("databaseType")
                sqlServerHostIP = rsx("databaseIPPath")
                sqlServerDatabaseName = rsx("databaseName")
                sqlServerUsername = rsx("databaseUser")
                sqlServerPassword = rsx("databasePass")

            else
                db_PREFIX=""
                databaseType = "access"
                MDBPath = phpTrim(rsx("databaseIPPath"))
                accessPass = rsx("databasePass") 
                if MDBPath="" then
                    call die("{""info"": ""Access数据库为空"": ""n""}")
                elseif checkFile(MDBPath)=false then
                    call die("{""info"": ""Access数据库文件不存在。"",""status"": ""n""}")
                elseif checkMdbPositionSafe(MDBPath)=false then
                    call die("{""info"": ""导入表到数据库失败，数据库位置不在/uploadfiles/mdb/里"",""pass"": """",""status"": ""n""}")
                end if


            end if
            call selectOpenConn(thisConn) '打开数据库'
            if databaseType="mysql" then
                fieldList=getMysqlFieldList_conn(rs("tablename"),thisConn)
            else
                fieldList=getFieldList_conn(rs("tablename"),thisConn)
            end if
            splstr=split(fieldList,",")
            'call echo("fieldList","fieldList")
            nOK=0
            sortrank=0
            for each s in splstr
                if s<>"" then
                    if databaseType="mysql" then
                        fieldAlt=getMysqlFieldAltNumb_conn(rs("tablename"), s,thisConn) 
                    else
                        fieldAlt=getFieldAltNumb_conn(rs("tablename"), s,thisConn) 
                    end if
                    if fieldAlt=3 and idIndexName="" then idIndexName=s
                    sortrank=sortrank+10
                    sql = "select * from ["& temp_db_PREFIX &  "layout_database_field] where layout_database_table_id="&rs("id") & " and fieldName='"&s&"'"
                    rss.open sql,conn,1,3
                    if rss.eof then  '在最后添加一条'
                        rss.addnew
                        rss("layout_database_table_id")=rs("id")
                        rss("layout_field_id")=find_layout_field_id(temp_db_PREFIX,s)
                        rss("fieldName")=s
                        rss("isthrough")=1
                        rss("sortrank")=sortrank
                        rss("fieldType")=fieldAlt   '字段数字类型'
                        rss.update
                        nOK=nOK+1
                    else
                        if rss("layout_field_id")=-1 then  '为-1,则再次查询对应的字段配置'
                            rss("layout_field_id")=find_layout_field_id(temp_db_PREFIX,s)
                            rss.update
                        end if
                    end if:rss.close

                end if
            next 
        end if:rsx.close

        if idIndexName<>"" then
            rs("idIndexName")=idIndexName
            rs.update
        end if

    end if:rs.close

    call die("{""info"": ""导入"&nOK&"字段到表"",""status"": ""y""}")  
end function
'从以有配置好的layout_field表里找对应的比较符合的'20250810
function find_layout_field_id(temp_db_PREFIX,fieldName)
    dim sql,layout_field_id
    layout_field_id=-1
    sql = "select * from ["& temp_db_PREFIX &  "layout_field] where fieldname='"&fieldName&"' and isthrough=1"
    ' call echo("sql",sql)
    rst.open sql,conn,1,1
    ' call echo("总数",rst.recordcount)
    while not rst.eof
        if layout_field_id=-1 then
            layout_field_id=rst("id")
        elseif rst("fieldType")="input" then
            layout_field_id=rst("id")
        end if 
    rst.movenext:wend:rst.close

    find_layout_field_id=layout_field_id
end function
