'统计用户对应的金额
function getMemberTotalMoney(rs)
    dim sql,rsx
    Set rsx = CreateObject("Adodb.RecordSet")
    sql="select sum(price)as totalMoney from "& getTableFieldFuHao(db_PREFIX&"money") &" where isthrough=1 and userid="&rs("id") 
    rsx.open sql,conn,1,1
    ' 返回总金额 
    getMemberTotalMoney=IIF(isnull(rsx("totalMoney"))=true,0,rsx("totalMoney")):rsx.close
end function
