'获得某会员去年总金额
function getMemberLastYearMoney(rs)
	dim sql,rsx,lastYearPrice
	Set rsx = CreateObject("Adodb.RecordSet")
	sql="select sum(price) as totalMoney from "& getTableFieldFuHao(db_PREFIX&"money") &" where isthrough=1 and userid="&rs("id")&" AND YEAR(daytime)=YEAR("&sqlAorS("now()")&")-1"
	rsx.open sql,conn,1,1
	lastYearPrice=IIF(isnull(rsx("totalMoney"))=true,0,rsx("totalMoney")):rsx.close
	getMemberLastYearMoney=lastYearPrice
end function
