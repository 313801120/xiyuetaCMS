'获得某会员前年总金额
function getMemberBeforeLastYearMoney(rs)
	dim sql,rsx,beforeLastYearPrice
	Set rsx = CreateObject("Adodb.RecordSet")
	sql="select sum(price) as totalMoney from "& getTableFieldFuHao(db_PREFIX&"money") &" where isthrough=1 and userid="&rs("id")&" AND YEAR(daytime)=YEAR("&sqlAorS("now()")&")-2"
	rsx.open sql,conn,1,1
	beforeLastYearPrice=IIF(isnull(rsx("totalMoney"))=true,0,rsx("totalMoney")):rsx.close
	getMemberBeforeLastYearMoney=beforeLastYearPrice
end function
