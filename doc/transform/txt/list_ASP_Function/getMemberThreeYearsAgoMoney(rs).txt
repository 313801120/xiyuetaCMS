'获得某会员大前年总金额
function getMemberThreeYearsAgoMoney(rs)
	dim sql,rsx,threeYearsAgoPrice
	Set rsx = CreateObject("Adodb.RecordSet")
	sql="select sum(price) as totalMoney from "& getTableFieldFuHao(db_PREFIX&"money") &" where isthrough=1 and userid="&rs("id")&" AND YEAR(daytime)=YEAR("&sqlAorS("now()")&")-3"
	rsx.open sql,conn,1,1
	threeYearsAgoPrice=IIF(isnull(rsx("totalMoney"))=true,0,rsx("totalMoney")):rsx.close
	getMemberThreeYearsAgoMoney=threeYearsAgoPrice
end function
