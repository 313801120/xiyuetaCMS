'解密等于号和逗号
function decDHJH(s)
	decDHJH=replace(replace(s,"[[{等于号}]]","=") ,"[[{逗号}]]", ",")
end function

function add_layout_table_action(action,layout_page_id,parentid,title,bgColor,noteIco,sortrank,customizeaction)

    dim rs:Set rs = CreateObject("Adodb.RecordSet")
    sql="select * from ["& db_PREFIX &"layout_table_action] where parentid="&parentid&" and layout_page_id="&layout_page_id&" and title='"&title&"'"
    rs.open sql,conn,1,3
    if rs.eof then

        rs("title")=title

        rs("bgColor")=bgColor

        rs("customizeaction")=customizeaction '自定义动作值'
        rs("itemWidth")=52

end function

'按菜单添加列表页里表单' 如  aa>>bb    20250904-2
function subAddLayout_listPage_form(action,layout_page_id,title,sortrank)   
    dim sql,parentid,splxx,s,nCount,isDebug,splA,buttontitle,splB,splStr
    isDebug=false
    dim rs:Set rs = CreateObject("Adodb.RecordSet")
    if isDebug then call echo("layout_page_id",layout_page_id)
    nCount=0
    splStr=split(title,">>")
    for each s in splStr
        s=phptrim(s)
        if s<>"" then
            splA=split(s&"[,]","[,]")
            s=splA(0)
            splB=split(splA(1)&"||","|")

            if parentid="" then
            ' call echo(action,s):doevents
                sql="select * from ["& db_PREFIX &"layout_listPage_form] where layout_page_id="&layout_page_id&" and parentid=-1 and title='"&s&"'"
                rs.open sql,conn,1,3
                if rs.eof then
                    rs.addnew
                    rs("layout_page_id")=layout_page_id
                    rs("parentid")=-1
                    rs("title")=s
                    splxx=split(s&"（","（")
                    buttontitle=splxx(0)
                    rs("buttontitle")=buttontitle
                    rs("isthrough")=1  
                    rs("sortrank")=sortrank 
                    if action="search" then   '搜索字段'
                        rs("saction")="fieldType"
                        rs("inputname")="searchType" 
                    elseif action="order" then   '排序选择
                        rs("saction")="orderSelect"
                        rs("inputname")="dataOrderyBy" 
                    elseif action="searchInput" then   '搜索input框'
                        rs("saction")="searchInput"
                        rs("inputname")="key"
                        rs("fieldname")="key"
                    elseif action="submit" then    '提交button按钮'
                        rs("saction")="submit"
                        rs("inputname")="submit" 
                    elseif action="time" then    '时间
                        rs("saction")="date"
                        rs("inputname")=splB(0)
                        rs("fieldname")=splB(1)
                        rs("datebjtype")=splB(2)
                    elseif action="line" then    '分割线'
                        rs("saction")="line"
                        rs("inputname")=splB(0) 
                    end if
                    rs.update:rs.close
                    nCount=nCount+1
                    rs.open sql,conn,1,1
                    if not rs.eof then
                        parentid=rs("id")
                    end if
                else
                    parentid=rs("id")
                end if:rs.close

                if isDebug then call echo("s",s):doevents
            else
                sql="select * from ["& db_PREFIX &"layout_listPage_form] where parentid="&parentid&" and title='"&s&"'"
                rs.open sql,conn,1,3
                if rs.eof then
                    rs.addnew
                    rs("layout_page_id")=layout_page_id
                    rs("parentid")=parentid
                    rs("title")=s
                    rs("buttontitle")=s
                    rs("isthrough")=1
                    if action="search" then
                        rs("defaultvalue")=s   '子类默认值'
                        rs("saction")="defaultValue"
                    elseif action="order" then
                        rs("defaultvalue")=s 
                        '处理标题显示值处理'
                        s=mid(s,10)
                        if right(s,5)=" desc" then
                            s=mid(s,1,len(s)-5) &"倒排序"
                        elseif right(s,4)=" asc" then
                            s=mid(s,1,len(s)-4) &"正排序"
                        else
                            s=s&"正排序"
                        end if 
                        ' rs("title")=s
                        rs("buttontitle")=s
                        rs("saction")="defaultValue"
                    end if
                    rs("sortrank")=sortrank  
                    rs.update
                    nCount=nCount+1
                end if:rs.close

                if isDebug then call echo("s",s):doevents
            end if
        end if
