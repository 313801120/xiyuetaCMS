function formatValue(val, indent) {
  indent = indent || 0;
  var spaces = "";
  for (var i = 0; i < indent; i++) {
    spaces += "  ";
  }
  
  if (val === null) {
    return "null";
  } else if (typeof val === "string") {
    return '"' + val + '"';
  } else if (typeof val === "number" || typeof val === "boolean") {
    return String(val);
  } else if (val instanceof Array) {
    var result = "[\n";
    for (var i = 0; i < val.length; i++) {
      result += spaces + "  " + formatValue(val[i], indent + 1);
      if (i < val.length - 1) result += ",";
      result += "\n";
    }
    result += spaces + "]";
    return result;
  } else if (typeof val === "object") {
    var result = "{\n";
    var keys = [];
    for (var key in val) {
      if (val.hasOwnProperty(key)) {
        keys.push(key);
      }
    }
    for (var i = 0; i < keys.length; i++) {
      var key = keys[i];
      result += spaces + "  \"" + key + "\": " + formatValue(val[key], indent + 1);
      if (i < keys.length - 1) result += ",";
      result += "\n";
    }
    result += spaces + "}";
    return result;
  }
  return String(val);
}
